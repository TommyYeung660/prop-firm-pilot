class TacticalExitDecision:
    action: Literal["HOLD", "MOVE_TO_BREAKEVEN", "TRAIL_SL", "REPRICE_TP", "PARTIAL_CLOSE", "EXIT_NOW"]
    state: Literal["INITIAL_RISK", "PROTECTION", "TREND_EXTENSION", "PROFIT_PROTECTION"]
    reason: str
    new_sl: float | None = None
    new_tp: float | None = None
    partial_close_ratio: float | None = None
    requires_llm_exception: bool = False


def choose_tactical_exit(
    snapshot: TacticalExitSnapshot,
    config: TacticalExitConfig,
) -> TacticalExitDecision:
    current_atr = compute_atr(snapshot.bars_1h, period=14) if not snapshot.bars_1h.empty else float("nan")
    ema_fast = compute_ema(snapshot.bars_5min["close"], 8).iloc[-1] if not snapshot.bars_5min.empty else None
    ema_slow = compute_ema(snapshot.bars_5min["close"], 21).iloc[-1] if not snapshot.bars_5min.empty else None
    rsi = compute_rsi(snapshot.bars_5min["close"], 14) if not snapshot.bars_5min.empty else None

    if snapshot.unrealized_r >= config.partial_close_min_r and _is_profit_protection(snapshot, ema_fast, ema_slow, rsi, current_atr):
        if not snapshot.partial_close_done:
            return TacticalExitDecision(
                action="PARTIAL_CLOSE",
                state="PROFIT_PROTECTION",
                reason="momentum_weakened",
                partial_close_ratio=config.partial_close_ratio,
            )

    if snapshot.unrealized_r >= config.breakeven_activation_r and _improves_stop(snapshot, snapshot.open_price):
        return TacticalExitDecision(
            action="MOVE_TO_BREAKEVEN",
            state="PROTECTION",
            reason="breakeven_threshold_reached",
            new_sl=snapshot.open_price,
        )

    return TacticalExitDecision(action="HOLD", state="INITIAL_RISK", reason="no_exit_signal")
```

@dataclass
class WriteBudgetSnapshot:
    write_remaining: int
    daily_write_limit: int

    @property
    def critical(self) -> bool:
        if self.daily_write_limit <= 0:
            return False
        return self.write_remaining < int(self.daily_write_limit * 0.15)


@dataclass
class TacticalExitEvaluation:
    decision: TacticalExitDecision
    skip_reason: str = ""
    requires_llm_exception_review: bool = False


class TacticalExitManager:
    def __init__(self, config: TacticalExitConfig) -> None:
        self._config = config

    def evaluate_position(
        self,
        snapshot: TacticalExitSnapshot,
        budget: WriteBudgetSnapshot,
        now: datetime,
    ) -> TacticalExitEvaluation:
        decision = choose_tactical_exit(snapshot, self._config)

        if self._blocked_by_cooldown(snapshot=snapshot, decision=decision, now=now):
            return TacticalExitEvaluation(
                decision=TacticalExitDecision(action="HOLD", state=decision.state, reason="cooldown_blocked"),
                skip_reason="modify_cooldown",
            )

        if budget.critical and decision.action not in {"EXIT_NOW", "MOVE_TO_BREAKEVEN"}:
            return TacticalExitEvaluation(
                decision=TacticalExitDecision(action="HOLD", state=decision.state, reason="write_budget_blocked"),
                skip_reason="write_budget_blocked",
            )

        return TacticalExitEvaluation(
            decision=decision,
            requires_llm_exception_review=decision.requires_llm_exception and self._config.use_llm_exception_path,
        )
```

Implementation notes:

- Keep this manager broker-agnostic: it should decide, not call MatchTrader directly

- Add a `tp_reprice_cooldown_seconds` branch separate from generic modify cooldown

class Scheduler:
    def __init__(...):
        ...
        self._tactical_exit_manager = TacticalExitManager(self._config.tactical.exit)

    async def _run_tactical_exit_cycle(
        self,
        open_positions: list[Any],
        opened_intents: list[TradeIntent],
    ) -> None:
        if not self._config.tactical.exit.enabled:
            return

        budget = WriteBudgetSnapshot(
            write_remaining=self._matchtrader.rate_limiter.write_remaining,
            daily_write_limit=self._matchtrader.rate_limiter.daily_write_limit,
        )
        intent_lookup = {
            intent.position_id: intent
            for intent in opened_intents
            if intent.position_id is not None
        }

        for pos in open_positions:
            intent = intent_lookup.get(str(pos.position_id))
            if intent is None:
                continue
            tactical_data = await self._fetch_tactical_data(intent.symbol)
            snapshot = self._build_tactical_exit_snapshot(pos, intent, tactical_data)
            evaluation = self._tactical_exit_manager.evaluate_position(snapshot, budget, self._now_utc())
            await self._handle_tactical_exit_evaluation(pos, intent, evaluation)

```python
async def _execute_tactical_exit_action(
    self,
    pos: Any,
    intent: TradeIntent,
    evaluation: TacticalExitEvaluation,
) -> None:
    decision = evaluation.decision
    if decision.action == "HOLD":
        self._log_trade_event("TACTICAL_EXIT_SKIPPED", {"position_id": str(pos.position_id), "reason": decision.reason})
        return

    if decision.action == "PARTIAL_CLOSE":
        close_volume = self._calculate_partial_close_volume(
            symbol=intent.symbol,
            current_volume=pos.volume,
            ratio=decision.partial_close_ratio or 0.0,
        )
        result = await self._matchtrader.close_position(
            position_id=str(pos.position_id),
            symbol=pos.symbol,
            side=pos.side,
            volume=close_volume,
        )
    else:
        result = await self._matchtrader.modify_position(
            position_id=str(pos.position_id),
            symbol=pos.symbol,
            side=pos.side,
            volume=pos.volume,
            sl=decision.new_sl if decision.action in {"MOVE_TO_BREAKEVEN", "TRAIL_SL"} else pos.sl_price,
            tp=decision.new_tp if decision.action == "REPRICE_TP" else pos.tp_price,
        )

    if not result.success:
        self._log_trade_event("TACTICAL_EXIT_SKIPPED", {"position_id": str(pos.position_id), "reason": result.message})
        return

    if decision.action in {"MOVE_TO_BREAKEVEN", "TRAIL_SL", "REPRICE_TP"}:
        verified = await self._matchtrader.verify_sl_tp(
            position_id=str(pos.position_id),
            expected_sl=decision.new_sl,
            expected_tp=decision.new_tp,
        )
        if not verified:
            self._log_trade_event("TACTICAL_EXIT_SKIPPED", {"position_id": str(pos.position_id), "reason": "verify_failed"})
            return

    self._update_tactical_exit_meta(intent, decision, pos)
    self._log_trade_event("TACTICAL_EXIT_ACTION", {"position_id": str(pos.position_id), "action": decision.action})
```
